\noindent \textbf{PART I: True or False [15 Points]}
\vspace{0.5em}

\noindent \textit{Write TRUE if the statement is unconditionally valid under standard set theory (ZFC), or FALSE if it is false or fails in general. No justification required for this section.}
    \begin{enumerate}[1.]
        \item{If $R$ is a symmetric and transitive relation on a non-empty set $A$, then $R$ must also be reflexive.}
       
        \item{Let $\Pi$ be a partition of a non-empty set $A$, and let $R$ be the equivalence relation induced by $\Pi$. If $x, y \in A$ belong to distinct blocks of $\Pi$, then $[x] \cap [y] = \emptyset$.}
        
        \item{If $f: A \to B$ and $g: B \to C$ are functions such that $g \circ f$ is surjective, then $f$ must be surjective.}
        
        \item{For any function $f: A \to B$ and any subsets $X, Y \subseteq A$, $f(X \cap Y) = f(X) \cap f(Y)$.}
        
        \item{Every finite partially ordered set $(A, \preceq)$ has at least one minimal element.}
        
        \item{If $(A, \preceq)$ is a partially ordered set in which every pair of elements has a greatest lower bound (infimum), then $(A, \preceq)$ is totally ordered.}
        
        \item{Every subset of a well-ordered set $(A, \preceq)$ is also well-ordered under the restricted ordering.}
        
        \item{If $A$ and $B$ are disjoint countable sets, then $A \cup B$ is denumerable.}
        
        \item{There exists a bijection between the open interval $(0, 1)$ and the closed interval $[0, 1]$.}
        
        \item{If $A$ is an uncountable set and $B$ is a subset of $A$ such that $B$ is uncountable, then $A \setminus B$ must be countable.}
        
        \item{The set of all polynomials with rational coefficients, $\mathbb{Q}[x]$, is denumerable.}
        
        \item{For any cardinal numbers $\alpha$ and $\beta$, if $\alpha < \beta$, then $2^\alpha < 2^\beta$.}
        
        \item{$\aleph_0^{\aleph_0} = \mathfrak{c}^{\aleph_0}$.}
        
        \item{If $\vert{}A\vert{} = \mathfrak{c}$, then the set of all finite subsets of $A$ has cardinality $\mathfrak{c}$.}
        
        \item{If $f: \mathbb{R} \to \mathbb{R}$ is strictly increasing, then $f$ is a bijection.}
    \end{enumerate}

\noindent \textbf{Part II. Computational \& Short-Answer Proofs [12 Points]}

\noindent \textit{Show complete, rigorous justifications for each item.}
\begin{enumerate}[1.]

    \item{[4 points] Simplify each cardinal expression to a single standard cardinal symbol ($\aleph_0$, $\mathfrak{c}$, or $2^{\mathfrak{c}}$). Justify each step using cardinal arithmetic laws.}
            \begin{enumerate}[(a)]
                \item{$(\aleph_0 \cdot \mathfrak{c} + 2^{\aleph_0})^{\aleph_0}$}
                \item{$\mathfrak{c}^{\mathfrak{c}} + 2^{\mathfrak{c}} \cdot \aleph_1$}
        \end{enumerate}
    
    \item{[4 points] Let $X = \{1, 2, 3, 4, 6, 12\}$ be ordered by divisibility ($a \preceq b \iff a \mid b$). Consider the subset $W = \{2, 3, 4\} \subset X$.}
            \begin{enumerate}[(a)]
                \item{Find all maximal elements of $W$ (within $W$ itself).}
                \item{Find all upper bounds of $W$ in $X$, and determine $\sup_X(W)$ if it exists.}
                \item{Find all lower bounds of $W$ in $X$, and determine $\inf_X(W)$ if it exists.}
                \item{Is $(W, \preceq)$ well-ordered? Justify your answer in one or two sentences.}
        \end{enumerate}
   
    \item{[4 points] Define the relation $\rho$ on $\mathbb{Z} \times (\mathbb{Z} \setminus \{0\})$ by:}
        \[(a, b) \rho (c, d) \iff ad = bc\]
            \begin{enumerate}[(a)]
                \item{Prove that $\rho$ is transitive. (You may assume reflexivity and symmetry hold).}
                \item{Describe the equivalence class $[(2, -3)]$ using set-builder notation.}
        \end{enumerate}        
\end{enumerate}

\noindent \textbf{Part III. Rigorous Proving [28 Points]}

\noindent \textit{Write formal, complete mathematical proofs. Clearly state your assumptions, definitions, and logical deductions.}
    \begin{enumerate}[1.]
        
        \item{[7 points] Let $A, B,$ and $C$ be non-empty sets, and let $f: A \to B$ and $g: B \to C$ be functions.}
            \begin{enumerate}[(a)]
                \item{[4 pts] Prove that if $g \circ f$ is surjective and $g$ is injective, then $f$ is surjective.}
                \item{[3 pts] Disprove the following statement by providing an explicit counterexample: "If $g \circ f$ is injective and $f$ is injective, then $g$ is injective."}
            \end{enumerate}
        
        \item{[7 points] Construct an explicit bijection $f: (0, 1] \to (0, 1)$ to prove that $(0, 1] \sim (0, 1)$. You must:}
            \begin{enumerate}[(a)]
                \item{Define $f(x)$ explicitly for all $x \in (0, 1]$ and verify that $f$ is well-defined (i.e., $f(x) \in (0, 1)$).}
                \item{Prove rigorously that $f$ is injective.}
                \item{Prove rigorously that $f$ is surjective.}
        \end{enumerate}
        
        \item{[7 points] Let $(A, \preceq)$ be a partially ordered set. Prove that if $A$ has a first (least) element $m$, then $m$ is the unique minimal element of $A$ (that is, $m$ is minimal, and no other element of $A$ is minimal).}

        \item{[7 points] Prove that the set of all functions from $\{0, 1\}$ to $\mathbb{N}$, denoted $\mathbb{N}^{\{0, 1\}}$, is denumerable by constructing an explicit bijection between $\mathbb{N}^{\{0, 1\}}$ and $\mathbb{N} \times \mathbb{N}$ (you may use the established fact from class/Quiz 11 that $\mathbb{N} \times \mathbb{N}$ is denumerable).}
    \end{enumerate}
